def map_network(s,x,y,w,h,prefix='地图网络'):
    rnd=random.Random(100)
    pts=[]
    for i in range(16):
        px=x+rnd.uniform(.05,.95)*w; py=y+rnd.uniform(.10,.90)*h; pts.append((px,py))
    # nearest connections
    for i,(px,py) in enumerate(pts):
        dists=sorted((( (px-qx)**2+(py-qy)**2,j,qx,qy) for j,(qx,qy) in enumerate(pts) if j!=i))[:2]
        for _,j,qx,qy in dists: line(s,px,py,qx,qy,'#0B6F9D',.7,trans=25,name=f'{prefix}_连接')
    for i,(px,py) in enumerate(pts): circle(s,px-.035,py-.035,.07,fill=YELLOW if i in [3,9,13] else CYAN,line=None,trans=5,name=f'{prefix}_节点{i}')
