
def bar_chart(s,x,y,w,h,prefix='柱状图',highlight=4):
    vals=[.35,.48,.58,.72,.86,.78,.64]
    bw=w/(len(vals)*1.55); gap=bw*.55
    for i,v in enumerate(vals):
        col=YELLOW if i==highlight else CYAN
        rect(s,x+i*(bw+gap),y+h-v*h,bw,v*h,fill=col,line=None,trans=0,radius=False,name=f'{prefix}_柱{i+1}')
    line(s,x-.05,y+h,x+w,y+h,'#4B7890',.8,name=f'{prefix}_基线')

def phone(s,x,y,w=1.2,h=2.0,prefix='手机'):
    rect(s,x,y,w,h,fill='#082B48',line=CYAN,lw=1,radius=True,name=f'{prefix}_外框')
    rect(s,x+.08,y+.18,w-.16,h-.38,fill='#0B3D5E',line='#1C7FAB',trans=10,lw=.6,radius=False,name=f'{prefix}_屏幕')
    for i in range(4):
        rect(s,x+.16,y+.38+i*.28,w-.32,.20,fill='#0D4B6C',line=None,trans=15,radius=True,name=f'{prefix}_菜单{i+1}')
    rect(s,x+.29,y+h-.38,w-.58,.18,fill='#138CCC',line=None,trans=0,radius=True,name=f'{prefix}_按钮')

def people_icon(s,cx,cy,z=.65,prefix='图标_队伍'):
    for i,dx in enumerate([-.22,0,.22]):
        circle(s,cx+dx*z-z*.08,cy-z*.30,z*.16,fill=CYAN if i!=1 else YELLOW,line=None,name=f'{prefix}_头{i+1}')
        rect(s,cx+dx*z-z*.13,cy-z*.06,z*.26,z*.30,fill='#0A4B70',line=CYAN,lw=.7,radius=True,name=f'{prefix}_身{i+1}')

def shield_icon(s,cx,cy,z=.60,prefix='图标_盾牌'):
    sh=s.shapes.add_shape(MSO_SHAPE.HEXAGON,Inches(cx-z*.36),Inches(cy-z*.40),Inches(z*.72),Inches(z*.80)); sh.name=f'{prefix}_盾'; sh.fill.solid(); sh.fill.fore_color.rgb=rgb('#0A4A70'); sh.line.color.rgb=rgb(CYAN); sh.line.width=Pt(1)
    line(s,cx-z*.14,cy,cx,cy+z*.13,YELLOW,1.5,name=f'{prefix}_勾1'); line(s,cx,cy+z*.13,cx+z*.19,cy-z*.14,YELLOW,1.5,name=f'{prefix}_勾2')

def coin_icon(s,cx,cy,z=.58,prefix='图标_基金'):
    for i in range(3): circle(s,cx-z*.30+i*z*.08,cy-z*.30+i*z*.05,z*.60,fill='#0B4C70',line=CYAN,lw=1,name=f'{prefix}_币{i+1}')
    txt(s,'¥',cx-z*.16,cy-z*.16,z*.32,z*.32,12,YELLOW,True,name=f'{prefix}_符号')

def clone_texts(src_slide,dst_slide):
    for sh in src_slide.shapes:
        if getattr(sh,'has_text_frame',False) and sh.text.strip():
            el=deepcopy(sh._element); dst_slide.shapes._spTree.insert_element_before(el,'p:extLst')
