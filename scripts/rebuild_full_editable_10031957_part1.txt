def rgb(h):
    h=h.replace('#',''); return RGBColor(int(h[:2],16),int(h[2:4],16),int(h[4:6],16))

def set_alpha_fill(shape, transparency):
    if shape.fill.type is None: return
    solid=shape.fill._xPr.find(qn('a:solidFill'))
    if solid is not None and len(solid):
        clr=solid[0]
        for a in list(clr.findall(qn('a:alpha'))): clr.remove(a)
        a=OxmlElement('a:alpha'); a.set('val',str(int((100-transparency)*1000))); clr.append(a)

def set_alpha_line(shape, transparency):
    ln=shape.line._get_or_add_ln(); sf=ln.find(qn('a:solidFill'))
    if sf is not None and len(sf):
        clr=sf[0]
        for a in list(clr.findall(qn('a:alpha'))): clr.remove(a)
        a=OxmlElement('a:alpha'); a.set('val',str(int((100-transparency)*1000))); clr.append(a)

def rect(s,x,y,w,h,fill=NAVY2,line=None,trans=0,lw=1,radius=False,name=None):
    typ=MSO_SHAPE.ROUNDED_RECTANGLE if radius else MSO_SHAPE.RECTANGLE
    sh=s.shapes.add_shape(typ,Inches(x),Inches(y),Inches(w),Inches(h))
    if name: sh.name=name
    sh.fill.solid(); sh.fill.fore_color.rgb=rgb(fill); set_alpha_fill(sh,trans)
    if line:
        sh.line.color.rgb=rgb(line); sh.line.width=Pt(lw)
    else: sh.line.fill.background()
    return sh

def circle(s,x,y,d,fill=NAVY2,line=CYAN,trans=0,lw=1,name=None):
    sh=s.shapes.add_shape(MSO_SHAPE.OVAL,Inches(x),Inches(y),Inches(d),Inches(d))
    if name: sh.name=name
    sh.fill.solid(); sh.fill.fore_color.rgb=rgb(fill); set_alpha_fill(sh,trans)
    if line:
        sh.line.color.rgb=rgb(line); sh.line.width=Pt(lw)
    else: sh.line.fill.background()
    return sh

def line(s,x1,y1,x2,y2,color=CYAN,width=1,trans=0,name=None,arrow=False):
    sh=s.shapes.add_connector(MSO_CONNECTOR.STRAIGHT,Inches(x1),Inches(y1),Inches(x2),Inches(y2))
    if name: sh.name=name
    sh.line.color.rgb=rgb(color); sh.line.width=Pt(width)
    if trans: set_alpha_line(sh,trans)
    if arrow: sh.line.end_arrowhead=True
    return sh

def txt(s,text,x,y,w,h,size=12,color=WHITE,bold=False,align=PP_ALIGN.CENTER,name=None):
    sh=s.shapes.add_textbox(Inches(x),Inches(y),Inches(w),Inches(h));
    if name: sh.name=name
    tf=sh.text_frame; tf.clear(); tf.word_wrap=True; tf.vertical_anchor=MSO_ANCHOR.MIDDLE
    tf.margin_left=tf.margin_right=tf.margin_top=tf.margin_bottom=0
    p=tf.paragraphs[0]; p.alignment=align
    r=p.add_run(); r.text=text; r.font.name=FONT_EN; r.font.size=Pt(size); r.font.bold=bold; r.font.color.rgb=rgb(color)
    rPr=r._r.get_or_add_rPr(); ea=rPr.find(qn('a:ea'))
    if ea is None: ea=OxmlElement('a:ea'); rPr.append(ea)
    ea.set('typeface',FONT_CN)
    return sh

def freeform_poly(s,points,fill,linec=None,trans=0,lw=.6,name=None):
    b=s.shapes.build_freeform(Inches(points[0][0]),Inches(points[0][1]),scale=1.0)
    verts=[(Inches(x),Inches(y)) for x,y in points[1:]]
    b.add_line_segments(verts,close=True)
    sh=b.convert_to_shape()
    if name: sh.name=name
    sh.fill.solid(); sh.fill.fore_color.rgb=rgb(fill); set_alpha_fill(sh,trans)
    if linec:
        sh.line.color.rgb=rgb(linec); sh.line.width=Pt(lw)
    else: sh.line.fill.background()
    return sh
