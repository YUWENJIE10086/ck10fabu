    p=doc.add_paragraph(); p.paragraph_format.line_spacing=1.5; p.paragraph_format.space_after=Pt(0)
    if indent: p.paragraph_format.first_line_indent=Cm(0.74)
    if not bold_phrases:
        r=p.add_run(text); set_run_font(r,BODY_FONT,Pt(size))
    else:
        # simple phrase splitting
        pattern='('+'|'.join(map(re.escape,sorted(bold_phrases,key=len,reverse=True)))+')'
        for part in re.split(pattern,text):
            if not part: continue
            r=p.add_run(part); set_run_font(r,BODY_FONT,Pt(size), part in bold_phrases)
    return p

def add_callout(doc,label,text):
    t=doc.add_table(rows=1, cols=1); t.alignment=WD_TABLE_ALIGNMENT.CENTER; t.autofit=True
    cell=t.cell(0,0); set_cell_shading(cell,'F2F2F2')
    p=cell.paragraphs[0]; p.paragraph_format.space_after=Pt(0); p.paragraph_format.left_indent=Cm(0.2); p.paragraph_format.right_indent=Cm(0.2)
    r=p.add_run(label+' '); set_run_font(r,HEAD_FONT,Pt(11),True)
    r=p.add_run(text); set_run_font(r,BODY_FONT,Pt(11))
    doc.add_paragraph().paragraph_format.space_after=Pt(0)

def add_script(doc, items):
    for typ,txt in items:
        if typ=='NOTE':
            add_callout(doc,'版本定位：',txt.replace('版本定位：',''))
        elif typ=='PPT':
            add_callout(doc,'【PPT】',txt)
        elif typ=='VIDEO':
            add_callout(doc,'【视频】',txt)
        elif typ in ('A','B'):
            p=doc.add_paragraph(); p.paragraph_format.line_spacing=1.45; p.paragraph_format.space_after=Pt(1)
            p.paragraph_format.left_indent=Cm(0.15)
            rr=p.add_run(f'{typ}：'); set_run_font(rr,HEAD_FONT,Pt(16),True)
            # bold signature phrases
            phrases=['两颗决策大脑','维修可以续命，但不能回春','有效烤能','区位熵','AHP','熵权法','TOPSIS','增长2%、5%、10%','先盘活、再修复、仍有缺口才新建','双模四步法','存量体检—增量推演—平台执行']
            remaining=txt
            pattern='('+'|'.join(map(re.escape,sorted([x for x in phrases if x in txt],key=len,reverse=True)))+')' if any(x in txt for x in phrases) else None
            if pattern:
                for part in re.split(pattern,txt):
                    if not part: continue
                    r=p.add_run(part); set_run_font(r,BODY_FONT,Pt(16),part in phrases)
            else:
                r=p.add_run(txt); set_run_font(r,BODY_FONT,Pt(16))
