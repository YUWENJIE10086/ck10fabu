export interface BarnListParams {
  pageNum?: number
  pageSize?: number
  county?: string
  township?: string
  keyword?: string
  status?: string
  healthLevel?: string
  facilityStatus?: string
}

export function getBarnList(params: BarnListParams = {}): Promise> {
  return get('/barn/project/list', params) as any
}

// src/api/dict.ts
export function getDictData(dictType: string): Promise {
  return get(`/system/dict/data/type/${dictType}`)
}

// 使用示例
const healthLevels = await getDictData('health_level')
const facilityStatuses = await getDictData('facility_status')
const repairUrgencies = await getDictData('repair_urgency')

⚠️ 注意事项

1. 数据同步：多个表之间存在关联关系，如 kf_basedata 和 ovens 通过 project_id 关联
2. 状态字段：大部分状态字段使用枚举值（如 status='baking'/'idle'）
3. 时间字段：所有时间字段统一使用 DATETIME 类型
4. 金额字段：金额字段使用 DECIMAL 类型，保留2位小数
5. 主键策略：部分表使用自增ID，部分表使用字符串ID
6. 字典管理：所有枚举值都应通过 sys_dict_data 表管理，便于维护
7. 分页查询：所有列表接口都支持分页，使用 pageNum 和 pageSize 参数
8. 搜索过滤：所有列表接口都支持按关键字、状态、类型等条件过滤
